% Required packages: tikz
% Required TikZ libraries: arrows.meta, graphs
% Target width: 160 mm maximum
% Minimum text size: 8 pt
% Colour/greyscale intent: colour and greyscale
% Directed edge shafts: 0.75 pt
% Dark blue bold text identifies an LLM call, redundantly with its operation name.
% Every arrow links two operations or a workflow boundary. The panel geometry
% describes adjacency and branching. Vertical gaps do not represent elapsed time.
\begin{tikzpicture}[
  x=1mm,y=1mm,
  title/.style={font=\fontsize{9.5}{11}\selectfont\bfseries,inner sep=0pt},
  vertex/.style={font=\fontsize{8.5}{10}\selectfont,inner xsep=2pt,inner ysep=2pt,align=center},
  llm/.style={vertex,font=\fontsize{8.5}{10}\selectfont\bfseries,text=blue!45!black},
  condition/.style={font=\fontsize{8}{9}\selectfont\itshape,inner sep=1pt},
  every edge/.style={draw=black!75,line width=.75pt,>={Stealth[length=4.8pt,width=3.4pt]}},
]
\path[use as bounding box] (0,-151) rectangle (156,3);

% All panels start at the top. Each compares structure at the same physical size.
\begin{scope}[xshift=26mm]
  \node[title] at (0,0) {(a) Single call};
  \node[vertex] (atask) at (0,-8) {Task};
  \node[llm] (adraft) at (0,-26) {Draft};
  \node[vertex] (aresult) at (0,-43) {Result};
  \graph[use existing nodes] {atask -> adraft -> aresult;};
\end{scope}

\begin{scope}[xshift=78mm]

\begin{scope}[xshift=130mm]
  \node[title] at (0,0) {(c) One repair};

    ctask -> cdraft -> creview;
    creview ->[edge node={node[condition,left] {clear}}] cresult;

\begin{scope}[xshift=130mm,yshift=-51mm]
